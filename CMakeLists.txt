cmake_minimum_required(VERSION 3.16)
project(deskport_core VERSION 0.1.0 LANGUAGES C CXX)
add_library(deskport_core INTERFACE)
target_include_directories(deskport_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(deskport_core INTERFACE c_std_11)
add_library(deskport_catalog INTERFACE)
target_include_directories(deskport_catalog INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/portable/include)
target_compile_features(deskport_catalog INTERFACE c_std_11)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME catalog COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/portable/test_catalog.py)
  add_test(NAME workspace COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_workspace.py)
endif()
